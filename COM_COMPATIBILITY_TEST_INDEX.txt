IMPLEMENTATION EXAMPLE: FALLBACK CASCADE
================================================================================

' VBScript implementation of fallback cascade
Function CreateCOMObjectRobust(progid, clsid)
    Dim obj
    On Error Resume Next
    
    ' Attempt 1: GetObject (existing instance)
    Set obj = GetObject(, progid)
    If Err.Number = 0 And Not IsEmpty(obj) Then
        CreateCOMObjectRobust = obj
        Exit Function
    End If
    
    ' Attempt 2: CreateObject with ProgID
    Set obj = CreateObject(progid)
    If Err.Number = 0 And Not IsEmpty(obj) Then
        CreateCOMObjectRobust = obj
        Exit Function
    End If
    
    ' Attempt 3: CreateObject with CLSID
    Set obj = CreateObject("CLSID:" & clsid)
    If Err.Number = 0 And Not IsEmpty(obj) Then
        CreateCOMObjectRobust = obj
        Exit Function
    End If
    
    ' All attempts failed
    CreateCOMObjectRobust = Nothing
    On Error GoTo 0
End Function
